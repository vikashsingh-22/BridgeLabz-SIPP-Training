Start
Define Enum TrafficLight
	Red
	Green
	Yellow
End Enum

Define Class TrafficSignal
PRIVATE currentLight: TrafficLight

CONSTRUCTOR TrafficSignal()
currentLight = TrafficLight.RED
END CONSTRUCTOR

METHOD changeLight()
      SWITCH (currentLight)
           CASE RED:
               currentLight = GREEN
           CASE GREEN:
               currentLight = YELLOW
           CASE YELLOW:
               currentLight = RED
        END SWITCH
    END METHOD

    METHOD displayLight()
        PRINT "Current Light "    
END METHOD
END CLASS

END
